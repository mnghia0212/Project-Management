Tasks {
  id: string,
  title: string,
  description: string,
  status: "todo" | "working" | "review" | "done",
  priority: "low" | "medium" | "high",
  dueDate: DateTime,
  createdAt: DateTime,
  updatedAt: DateTime,
  completedOn: DateTime
  department: string (Options),     
  assigneeIds: string Member Ids[],     
  sharedToIds: string Member Ids[], 
  creatorId: string,         
  attachments: string[],    
  comments: string Comment Ids[], 
  childTaskIds: string Task Ids[],
}

  avatarUrl?: string,

}
